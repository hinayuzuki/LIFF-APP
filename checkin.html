function doPost(e) { try { const raw = e.postData && e.postData.contents ? e.postData.contents : "{}"; const data = JSON.parse(raw); const lineId = data.lineId; const box = Number(data.box); const ss = SpreadsheetApp.openById("1J_FkVaRYs0quvBnZZaLH-Inj_apP5Pz-jcmkh810Fw0"); const sheet = ss.getSheetByName("受付"); if (!sheet) { return ContentService.createTextOutput("ERROR: シートが見つかりません"); } const lastRow = sheet.getLastRow(); const values = sheet.getRange(2, 1, lastRow - 1, 15).getValues(); for (let i = 0; i < values.length; i++) { const row = values[i]; const rowLineId = row[2]; const status = row[14]; // 0=未チェックイン, 1=チェックイン済み // ★ すでにチェックイン済みの場合 if (rowLineId === lineId && status === 1) { return ContentService.createTextOutput("CHECKED"); } // ★ 未チェックイン → チェックイン処理 if (rowLineId === lineId && status === 0) { const targetRow = i + 2; sheet.getRange(targetRow, 13).setValue(box); sheet.getRange(targetRow, 15).setValue(1); return ContentService.createTextOutput("OK"); } } return ContentService.createTextOutput("NO_MATCH"); } catch (err) { return ContentService.createTextOutput("ERROR: " + err); } }
