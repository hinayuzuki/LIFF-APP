// 10秒待つ関数 function wait(ms) { return new Promise(resolve => setTimeout(resolve, ms)); } async function main() { try { log("1: LIFF 初期化開始"); await liff.init({ liffId: "2010026333-wO6tpBt7" }); log("2: LIFF 初期化成功"); if (!liff.isLoggedIn()) { log("3: 未ログイン → ログインへ"); liff.login(); return; } log("4: プロフィール取得開始"); const profile = await liff.getProfile(); const userId = profile.userId; log("5: userId = " + userId); const apiUrl = "https://script.google.com/macros/s/AKfycbxWxfIjmHtYgc8h2FSTWIE2vTIAUTdvjYvnyJfteZfRyx0So494y0NmDpdHsb9ppQg/exec?lineId=" + userId; log("6: fetch 開始 → " + apiUrl); const res = await fetch(apiUrl); const formUrl = await res.text(); log("7: doGet 返り値 = " + formUrl); // ★ ここで10秒待つ log("8: 10秒待機開始"); await wait(10000); log("9: 10秒待機終了 → 遷移開始"); window.location.href = encodeURI(formUrl); } catch (e) { log("エラー: " + e); } }
